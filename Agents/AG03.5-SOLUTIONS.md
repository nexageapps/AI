## Exercise 1: Banking Agent

**Requirements:**
- Tools for checking balance, transferring funds, viewing history
- Account validation
- Sufficient funds check
- Rate limiting (10 transfers/min)
- Metrics tracking
- Error handling

### Solution

```python
from typing import List, Dict, Optional
from datetime import datetime, timedelta
from pydantic import BaseModel, Field, validator
from collections import deque
import json

# Models
class Account(BaseModel):
    id: int
    owner: str
    balance: float = Field(ge=0)
    transactions: List[Dict] = []

class Transaction(BaseModel):
    id: int
    from_account: int
    to_account: int
    amount: float
    timestamp: datetime
    status: str

# Database (simulated)
accounts_db = {
    1001: Account(id=1001, owner="Alice", balance=5000.00),
    1002: Account(id=1002, owner="Bob", balance=2500.00),
    1003: Account(id=1003, owner="Charlie", balance=10000.00),
}

transactions_db: List[Transaction] = []
transaction_id_counter = 5000

# Rate limiter for transfers
transfer_limiter = RateLimiter(max_calls=10, time_window=60)

# Custom exceptions
class AccountNotFoundError(ToolError):
    pass

class InsufficientFundsError(ToolError):
    pass

class TransferLimitExceededError(ToolError):
    pass

# Tools
@tool
@instrumented_tool
@safe_tool_execution
def check_balance(account_id: int) -> Dict[str, any]:
    """Check account balance.
    
    Args:
        account_id: Account ID to check
        
    Returns:
        Account balance and owner information
        
    Raises:
        AccountNotFoundError: If account doesn't exist
    """
    if account_id not in accounts_db:
        raise AccountNotFoundError(
            f"Account {account_id} not found. Valid accounts: {list(accounts_db.keys())}"
        )
    
    account = accounts_db[account_id]
    return {
        "account_id": account.id,
        "owner": account.owner,
        "balance": account.balance,
        "currency": "USD"
    }


@tool
@instrumented_tool
@safe_tool_execution
def transfer_funds(
    from_account: int,
    to_account: int,
    amount: float,
    description: str = "Transfer"
) -> Dict[str, any]:
    """Transfer funds between accounts.
    
    Args:
        from_account: Source account ID
        to_account: Destination account ID
        amount: Amount to transfer (must be positive)
        description: Optional transfer description
        
    Returns:
        Transfer confirmation with transaction ID
        
    Raises:
        AccountNotFoundError: If either account doesn't exist
        InsufficientFundsError: If source account lacks funds
        TransferLimitExceededError: If rate limit exceeded
        ValidationError: If amount is invalid
    """
    global transaction_id_counter
    
    # Validate amount
    if amount <= 0:
        raise ValidationError(f"Amount must be positive, got {amount}")
    
    # Check rate limit
    if not transfer_limiter.allow_call():
        raise TransferLimitExceededError(
            "Transfer rate limit exceeded (10 per minute). Please wait."
        )
    
    # Validate accounts exist
    if from_account not in accounts_db:
        raise AccountNotFoundError(f"Source account {from_account} not found")
    
    if to_account not in accounts_db:
        raise AccountNotFoundError(f"Destination account {to_account} not found")
    
    if from_account == to_account:
        raise ValidationError("Cannot transfer to the same account")
    
    # Check sufficient funds
    source_account = accounts_db[from_account]
    if source_account.balance < amount:
        raise InsufficientFundsError(
            f"Insufficient funds. Available: ${source_account.balance:.2f}, "
            f"Requested: ${amount:.2f}"
        )
    
    # Execute transfer
    dest_account = accounts_db[to_account]
    source_account.balance -= amount
    dest_account.balance += amount
    
    # Record transaction
    transaction = Transaction(
        id=transaction_id_counter,
        from_account=from_account,
        to_account=to_account,
        amount=amount,
        timestamp=datetime.now(),
        status="completed"
    )
    transactions_db.append(transaction)
    transaction_id_counter += 1
    
    # Add to account histories
    tx_record = {
        "id": transaction.id,
        "type": "debit",
        "amount": amount,
        "to": to_account,
        "timestamp": transaction.timestamp.isoformat(),
        "description": description
    }
    source_account.transactions.append(tx_record)
    
    tx_record_dest = {
        "id": transaction.id,
        "type": "credit",
        "amount": amount,
        "from": from_account,
        "timestamp": transaction.timestamp.isoformat(),
        "description": description
    }
    dest_account.transactions.append(tx_record_dest)
    
    return {
        "transaction_id": transaction.id,
        "from_account": from_account,
        "to_account": to_account,
        "amount": amount,
        "new_balance": source_account.balance,
        "status": "completed",
        "timestamp": transaction.timestamp.isoformat()
    }


@tool
@instrumented_tool
@safe_tool_execution
def get_transaction_history(
    account_id: int,
    days: int = 30,
    transaction_type: Optional[Literal["credit", "debit", "all"]] = "all"
) -> Dict[str, any]:
    """Get transaction history for an account.
    
    Args:
        account_id: Account ID to query
        days: Number of days of history (default: 30)
        transaction_type: Filter by transaction type
        
    Returns:
        List of transactions within the time period
        
    Raises:
        AccountNotFoundError: If account doesn't exist
        ValidationError: If days is invalid
    """
    if account_id not in accounts_db:
        raise AccountNotFoundError(f"Account {account_id} not found")
    
    if days <= 0 or days > 365:
        raise ValidationError("Days must be between 1 and 365")
    
    account = accounts_db[account_id]
    cutoff_date = datetime.now() - timedelta(days=days)
    
    # Filter transactions
    transactions = account.transactions
    
    # Filter by date
    recent_transactions = [
        tx for tx in transactions
        if datetime.fromisoformat(tx["timestamp"]) >= cutoff_date
    ]
    
    # Filter by type
    if transaction_type != "all":
        recent_transactions = [
            tx for tx in recent_transactions
            if tx["type"] == transaction_type
        ]
    
    return {
        "account_id": account_id,
        "owner": account.owner,
        "period_days": days,
        "transaction_type": transaction_type,
        "count": len(recent_transactions),
        "transactions": recent_transactions
    }


# Test the banking agent
def test_banking_agent():

    # Test 4: View history

    result = get_transaction_history(1001, days=30)

    # Test 5: Error handling - insufficient funds
    print("\n5. Test insufficient funds error:")
    try:
        result = transfer_funds(1002, 1001, 10000.00)
    except Exception as e:
        print(f"Expected error: {e}")
    
    # Test 6: Error handling - invalid account
    print("\n6. Test invalid account error:")
    try:
        result = check_balance(9999)
    except Exception as e:
        print(f"Expected error: {e}")
    
    # Test 7: Rate limiting
    print("\n7. Test rate limiting (attempting 12 transfers):")

    # Print metrics
```

## Exercise 2: Debug Buggy Tool

**Original (buggy) code:**
```python
@tool
def buggy_booking_tool(date: str, time: str, guests: int) -> str:
    """Book a restaurant table."""
    return f"Booked for {date} at {time} for {guests} guests"
```

### Issues Identified

1. No date validation (accepts any string)
2. No time validation (accepts invalid times)
3. No guest count validation (could be negative)
4. No availability check
5. No error handling
6. Returns unstructured string (not JSON)
7. No booking ID for tracking
8. No confirmation of success

### Solution

```python
from datetime import datetime, date, time as time_module
from pydantic import BaseModel, Field, validator
from typing import Literal, Optional
import json

class RestaurantBooking(BaseModel):
    """Restaurant booking data model."""
    booking_id: int
    date: date
    time: time_module
    guests: int
    status: Literal["confirmed", "pending", "cancelled"]
    table_number: Optional[int] = None
    customer_name: str

# Simulated database
bookings_db: List[RestaurantBooking] = []
booking_id_counter = 1000

# Restaurant configuration
OPENING_TIME = time_module(11, 0)  # 11:00 AM
CLOSING_TIME = time_module(22, 0)  # 10:00 PM
MAX_GUESTS_PER_BOOKING = 10
TABLES_AVAILABLE = 15

class BookingRequest(BaseModel):
    """Validated booking request."""
    date: date = Field(description="Booking date in YYYY-MM-DD format")
    time: str = Field(description="Booking time in HH:MM format (24-hour)")
    guests: int = Field(ge=1, le=MAX_GUESTS_PER_BOOKING, description=f"Number of guests (1-{MAX_GUESTS_PER_BOOKING})")
    customer_name: str = Field(min_length=2, max_length=100, description="Customer name")
    
    @validator('date')
    def validate_date(cls, v):
        """Validate booking date is in the future."""
        if v < date.today():
            raise ValueError(f"Cannot book in the past. Earliest date: {date.today()}")
        
        # Can't book more than 90 days in advance
        max_date = date.today() + timedelta(days=90)
        if v > max_date:
            raise ValueError(f"Cannot book more than 90 days in advance. Latest date: {max_date}")
        
        return v
    
    @validator('time')
    def validate_time(cls, v):
        """Validate time format and restaurant hours."""
        try:
            booking_time = datetime.strptime(v, "%H:%M").time()
        except ValueError:
            raise ValueError(f"Invalid time format. Use HH:MM (e.g., '19:30')")
        
        if booking_time < OPENING_TIME or booking_time > CLOSING_TIME:
            raise ValueError(
                f"Restaurant is open {OPENING_TIME.strftime('%H:%M')}-{CLOSING_TIME.strftime('%H:%M')}. "
                f"Requested: {v}"
            )
        
        return v


def check_availability(booking_date: date, booking_time: time_module) -> bool:
    """Check if tables are available at requested date/time."""
    # Count bookings at the same time slot (within 2 hour window)
    booking_datetime = datetime.combine(booking_date, booking_time)
    
    conflicting_bookings = [
        b for b in bookings_db
        if b.date == booking_date
        and b.status == "confirmed"
        and abs((datetime.combine(b.date, b.time) - booking_datetime).total_seconds()) < 7200  # 2 hours
    ]
    
    return len(conflicting_bookings) < TABLES_AVAILABLE


@tool
@safe_tool_execution
@instrumented_tool
def book_restaurant_table(
    date: str,
    time: str,
    guests: int,
    customer_name: str
) -> Dict[str, any]:
    """Book a restaurant table with full validation.
    
    Args:
        date: Booking date (YYYY-MM-DD format, future dates only)
        time: Booking time (HH:MM format, 24-hour, must be during business hours)
        guests: Number of guests (1-10)
        customer_name: Customer name for the booking
        
    Returns:
        Booking confirmation with booking ID and table number
        
    Raises:
        ValidationError: If any parameter is invalid
        ResourceNotFoundError: If no tables available
    """
    global booking_id_counter
    
    # Parse and validate inputs using Pydantic
    try:
        request = BookingRequest(
            date=datetime.strptime(date, "%Y-%m-%d").date(),
            time=time,
            guests=guests,
            customer_name=customer_name
        )
    except ValueError as e:
        raise ValidationError(f"Validation failed: {str(e)}")
    
    # Parse time for availability check
    booking_time = datetime.strptime(time, "%H:%M").time()
    
    # Check availability
    if not check_availability(request.date, booking_time):
        raise ResourceNotFoundError(
            f"No tables available on {date} at {time}. "
            f"Please try a different time or call the restaurant."
        )
    
    # Create booking
    booking = RestaurantBooking(
        booking_id=booking_id_counter,
        date=request.date,
        time=booking_time,
        guests=request.guests,
        status="confirmed",
        table_number=len(bookings_db) % TABLES_AVAILABLE + 1,  # Simplified assignment
        customer_name=request.customer_name
    )
    
    bookings_db.append(booking)
    booking_id_counter += 1
    
    return {
        "booking_id": booking.booking_id,
        "status": "confirmed",
        "date": booking.date.isoformat(),
        "time": booking.time.strftime("%H:%M"),
        "guests": booking.guests,
        "table_number": booking.table_number,
        "customer_name": booking.customer_name,
        "restaurant_phone": "555-0123",
        "message": f"Booking confirmed! Table {booking.table_number} reserved for {guests} guests."
    }


@tool
@safe_tool_execution
def cancel_booking(booking_id: int) -> Dict[str, any]:
    """Cancel a restaurant booking.
    
    Args:
        booking_id: Booking ID to cancel
        
    Returns:
        Cancellation confirmation
    """
    booking = next((b for b in bookings_db if b.booking_id == booking_id), None)
    
    if not booking:
        raise ResourceNotFoundError(f"Booking {booking_id} not found")
    
    if booking.status == "cancelled":
        return {
            "booking_id": booking_id,
            "status": "already_cancelled",
            "message": "This booking was already cancelled"
        }
    
    booking.status = "cancelled"
    
    return {
        "booking_id": booking_id,
        "status": "cancelled",
        "date": booking.date.isoformat(),
        "time": booking.time.strftime("%H:%M"),
        "message": f"Booking #{booking_id} cancelled successfully"
    }


# Tests
def test_booking_system():
    """Test the fixed booking system."""
    print("=== Restaurant Booking System Tests ===\n")
    
    # Test 1: Valid booking
    print("1. Valid booking:")
    tomorrow = (date.today() + timedelta(days=1)).isoformat()
    result = book_restaurant_table(tomorrow, "19:30", 4, "John Doe")
    print(json.dumps(result, indent=2))
    
    # Test 2: Invalid date (past)
    print("\n2. Test past date error:")
    result = book_restaurant_table("2020-01-01", "19:00", 2, "Jane Smith")
    print(json.dumps(result, indent=2))
    
    # Test 3: Invalid time format
    print("\n3. Test invalid time format:")
    result = book_restaurant_table(tomorrow, "7:30 PM", 2, "Bob Johnson")
    print(json.dumps(result, indent=2))
    
    # Test 4: Outside business hours
    print("\n4. Test outside business hours:")
    result = book_restaurant_table(tomorrow, "23:00", 2, "Alice Brown")
    print(json.dumps(result, indent=2))
    
    # Test 5: Too many guests
    print("\n5. Test too many guests:")
    result = book_restaurant_table(tomorrow, "19:00", 50, "Big Party")
    print(json.dumps(result, indent=2))
    
    # Test 6: Cancel booking
    print("\n6. Cancel booking:")
    result = cancel_booking(1000)
    print(json.dumps(result, indent=2))

if __name__ == "__main__":
    test_booking_system()
```

### Improvements Made

1. ✅ **Date Validation:** Checks format, future dates only, max 90 days advance
2. ✅ **Time Validation:** 24-hour format, within business hours
3. ✅ **Guest Validation:** 1-10 guests using Pydantic constraints
4. ✅ **Availability Check:** Prevents double-booking
5. ✅ **Error Handling:** Structured errors with clear messages
6. ✅ **Structured Output:** Returns JSON with all booking details
7. ✅ **Booking ID:** Unique ID for tracking
8. ✅ **Cancellation:** Added cancel functionality
9. ✅ **Metrics:** Instrumentation for monitoring

## Exercise 3: Optimize Parallel Execution

**Original (sequential):**

### Solution: Two Approaches

#### Approach 1: LLM-Driven Parallel Execution

The LLM automatically calls tools in parallel if they're independent. The key is prompt design.

#### Approach 2: Explicit Aggregator Tool

If you want guaranteed parallel execution, create an aggregator tool.

```python
import concurrent.futures
import time
import json

def fetch_user_profile(user_id: int) -> dict:
    """Internal function for profile fetch."""
    time.sleep(1)
    return {"name": "Alice", "email": "alice@example.com"}

def fetch_user_orders(user_id: int) -> dict:
    """Internal function for orders fetch."""
    time.sleep(1)
    return {"orders": [{"id": 1001, "total": 299.99}]}

def fetch_user_reviews(user_id: int) -> dict:
    """Internal function for reviews fetch."""
    time.sleep(1)
    return {"reviews": [{"product": 101, "rating": 5}]}


@tool
def get_complete_user_data(user_id: int) -> str:
    """Get all user data (profile, orders, reviews) in parallel.
    
    Args:
        user_id: User ID to fetch data for
        
    Returns:
        Complete user data from all sources
    """
    start = time.time()
    
    # Execute all fetches in parallel
    with concurrent.futures.ThreadPoolExecutor(max_workers=3) as executor:
        future_profile = executor.submit(fetch_user_profile, user_id)
        future_orders = executor.submit(fetch_user_orders, user_id)
        future_reviews = executor.submit(fetch_user_reviews, user_id)
        
        # Wait for all to complete
        profile = future_profile.result()
        orders = future_orders.result()
        reviews = future_reviews.result()
    
    elapsed = time.time() - start
    
    result = {
        "user_id": user_id,
        "profile": profile,
        "orders": orders,
        "reviews": reviews,
        "fetch_time_seconds": round(elapsed, 2)
    }
    
    return json.dumps(result, indent=2)


# Test
print("Fetching complete user data...")
start = time.time()
result = get_complete_user_data(100)
elapsed = time.time() - start

print(f"Result:\n{result}")
print(f"\nTotal time: {elapsed:.2f}s (expect ~1s, not ~3s)")
```

### Key Insights

**Prompt Engineering for Parallel Execution:**
- ✅ Use "simultaneously", "all at once", "in parallel"
- ✅ Avoid "then", "after that", "first"
- ✅ Group independent operations in one sentence

**Tool Design for Parallel Execution:**
- ✅ Make tools side-effect free (no state modification)
- ✅ Tools should not depend on each other
- ✅ Return structured JSON for easy aggregation

**Performance Comparison:**
```
Sequential:  3.0s (1s + 1s + 1s)
Parallel:    1.0s (max(1s, 1s, 1s))
Improvement: 3x faster
```
